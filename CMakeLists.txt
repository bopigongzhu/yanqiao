cmake_minimum_required(VERSION 3.24)
project(EShuRu VERSION 0.2.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MINGW)
  add_compile_options(-Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
  add_link_options(-static)
endif()
include_directories(src third_party)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(CMAKE_RC_FLAGS "${CMAKE_RC_FLAGS} -I${CMAKE_SOURCE_DIR}/src -F pe-x86-64")
else()
  set(CMAKE_RC_FLAGS "${CMAKE_RC_FLAGS} -I${CMAKE_SOURCE_DIR}/src -F pe-i386")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(shared STATIC src/common.cpp src/ipc.cpp src/registration.cpp)
target_link_libraries(shared PUBLIC advapi32 ole32 shell32 uuid)
add_executable(EShuRuMaintenance src/maintenance.cpp src/app.rc)
target_link_libraries(EShuRuMaintenance PRIVATE shared imm32)
add_library(engine STATIC src/core.cpp src/rime_engine.cpp src/ai.cpp src/frequency.cpp)
target_link_libraries(engine PUBLIC shared winhttp crypt32)
add_executable(api_probe EXCLUDE_FROM_ALL tests/api_probe.cpp)
target_link_libraries(api_probe PRIVATE engine)
add_executable(EShuRu WIN32 src/main.cpp src/panel.cpp src/app.cpp src/sandbox.cpp src/app.rc)
target_link_libraries(EShuRu PRIVATE engine d2d1 dwrite dwmapi comctl32 winmm psapi imm32 shcore)
target_link_options(EShuRu PRIVATE -municode)
add_library(EShuRuTip SHARED src/tsf.cpp src/tsf.def src/app.rc)
set_target_properties(EShuRuTip PROPERTIES PREFIX "")
target_link_libraries(EShuRuTip PRIVATE shared)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE engine psapi)
add_executable(integration_tests tests/integration_tests.cpp)
target_link_libraries(integration_tests PRIVATE engine psapi imm32)
add_executable(tsf_smoke tests/tsf_smoke.cpp)
target_link_libraries(tsf_smoke PRIVATE shared)
add_executable(ipc_cancel_tests tests/ipc_cancel_tests.cpp)
target_link_libraries(ipc_cancel_tests PRIVATE shared)
enable_testing()
add_executable(panel_tests tests/panel_tests.cpp src/panel.cpp)
target_link_libraries(panel_tests PRIVATE engine d2d1 dwrite dwmapi comctl32 shcore imm32)
add_test(NAME panel_dpi COMMAND panel_tests)
set_tests_properties(panel_dpi PROPERTIES TIMEOUT 30 WORKING_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
add_executable(frequency_tests tests/frequency_tests.cpp)
target_link_libraries(frequency_tests PRIVATE engine psapi)
add_test(NAME frequency COMMAND frequency_tests)
set_tests_properties(frequency PROPERTIES TIMEOUT 30 WORKING_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
add_executable(paging_tests tests/paging_tests.cpp)
target_link_libraries(paging_tests PRIVATE engine)
add_test(NAME paging COMMAND paging_tests)
add_executable(frequency_storage_tests tests/frequency_storage_tests.cpp)
target_link_libraries(frequency_storage_tests PRIVATE engine)
add_test(NAME frequency_storage COMMAND frequency_storage_tests)
set_tests_properties(paging frequency_storage PROPERTIES TIMEOUT 60 WORKING_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
add_test(NAME core COMMAND core_tests)
add_test(NAME tsf_lifecycle COMMAND tsf_smoke)
add_test(NAME ipc_cancel COMMAND ipc_cancel_tests)
set_tests_properties(ipc_cancel PROPERTIES TIMEOUT 15)
set_tests_properties(core PROPERTIES WORKING_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
set_tests_properties(tsf_lifecycle PROPERTIES WORKING_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
